def median(xs):
    s = sorted(xs)
    n = len(s)
    if n % 2 == 1:
        return s[n // 2]
    return (s[n // 2 - 1] + s[n // 2]) / 2

## interleave
def interleave(a, b):
    out = []
    for i in range(max(len(a), len(b))):
        if i < len(a):
            out.append(a[i])
        if i < len(b):
            out.append(b[i])
    return out

def caesar_shift(s, k):
    out = []
    for c in s:
        if 'a' <= c <= 'z':
            out.append(chr((ord(c) - ord('a') + k) % 26 + ord('a')))

## compress_runs
def compress_runs(s):
    if not s:
        return ''
    out = []
    cur = s[0]
    n = 1
    for c in s[1:]:
        if c == cur:
            n += 1
        else:
            out.append(cur + str(n))
            cur = c
            n = 1
    out.append(cur + str(n))
    return ''.join(out)
